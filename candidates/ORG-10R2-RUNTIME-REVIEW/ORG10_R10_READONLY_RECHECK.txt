    function Assert-Tree($Directory,$Expected) {
        $RootItem = Get-Item -LiteralPath $Directory -Force
        if (!$RootItem.PSIsContainer -or ($RootItem.Attributes -band [IO.FileAttributes]::ReparsePoint)) { throw 'ROOT_HOLD' }

    function Test-Zip($File,$Directory,$Expected) {
        $Zip = [IO.Compression.ZipFile]::OpenRead($File)
        $Count = 0
        try {
            foreach ($Entry in $Zip.Entries) {
                if ([string]::IsNullOrEmpty($Entry.Name)) { continue }
                $Stream = $Entry.Open()
                $Sha = [Security.Cryptography.SHA256]::Create()
                try { $Hash = [BitConverter]::ToString($Sha.ComputeHash($Stream)).Replace('-','').ToLowerInvariant() }
                finally { $Stream.Dispose(); $Sha.Dispose() }
                Assert-Hash (Join-Path $Directory ($Entry.FullName.Replace('/','\'))) $Hash
                $Count++
            }
        } finally { $Zip.Dispose() }
        if ($Count -ne $Expected) { throw 'ZIP_COUNT_HOLD' }
        Write-Host "$(Split-Path $File -Leaf): $Count/$Expected PASS"
    }
